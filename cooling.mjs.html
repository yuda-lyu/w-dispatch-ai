import isobj from 'wsemi/src/isobj.mjs'
import isfun from 'wsemi/src/isfun.mjs'
import isestr from 'wsemi/src/isestr.mjs'
import ispint from 'wsemi/src/ispint.mjs'
import cint from 'wsemi/src/cint.mjs'
import isenum from 'wsemi/src/isenum.mjs'
import { callHookSync } from './callUserFn.mjs'


// cooling.mjs — 條目冷卻紀錄、冷卻加長與持久連敗(單一擁有者; 2026-11-11 da402／di402, 併 Gitea #130; 帳本 R-COOL-2、R-COOL-3)
//
// 【為何集中於此】條目冷卻(state.cooling)曾有三個寫者(遞補層、知識管線之健康層、免費池)各寫一個時間戳, 讀者讀到過期即刪紀錄,
//   任何「次數」都存不下, 冷卻視窗只能固定(#130: 每小時一輪之呼叫端, 15分鐘視窗到下一輪已過期, 掛起之條目每輪照等一次)。
//   本檔為冷卻紀錄之唯一寫法: 遞補層之內建觸發與呼叫端(經匯出之coolEntry)同一規則, 讀取經coolingOf／streakOf。
//
// 【紀錄】state.cooling[id] = { at, n, reason }:
//   at     最近一次觸發之時刻(毫秒); 晚於現在者視為現在(防時鐘回撥, 讀到即改寫為現在);
//   n      加長之級數(≥1); 只在「視窗外」之觸發+1, 視窗內之觸發只更新at(所有寫者一律如此)——同一次呼叫內第二次觸發必在視窗內, 故每次呼叫每條目至多+1;
//   reason 最近一次改變n或建立紀錄之原因: 'timeout'(逾時)、'rate-limit'(限流: HTTP 429或failureKind rate-limit)、'detect'(coolDetect判定)、
//          'streak'(持久連敗達門檻)、'quota'(整組沒額度或呼叫端之額度池用盡)、'budget'(鏈首被呼叫端預算封頂, 見dispatchAiFallback【封頂逾時】);
//          'quota'與'budget'為「固定視窗原因」: 不動n(新建之紀錄n為1), 有效視窗恆為cooldownMs——額度用盡逐日發生, 若加長會逐日累加n(S6);
//          被預算封頂不是條目之故障, 只降序一個固定視窗以免結構性小預算下遞補永遠輪不到。
//   有效視窗＝固定視窗原因 ? cooldownMs : min(cooldownMs × cooldownGrowth^(n−1), cooldownMaxMs)。cooldownGrowth預設1(視窗恆為cooldownMs, 行為同修前)。
//   renewed(2026-11-11 排程第421項): 每次寫入另算「寫入前是否已在有效視窗內」——視窗內之重申(只更新at, n與reason不變)為true, 新建或視窗外之新觸發為false;
//   applyCool回傳、coolEntry回傳與遞補層條目層cooled事件皆帶此欄, 令只觀察之呼叫端不必自記本行程之事件(跨行程不可得)即可分辨重申與新觸發。
//   成交刪紀錄(n歸零); 讀到過期不刪(保留n, 下次視窗外之觸發才能加長); 掃整張表清除「現在−at ≥ 有效視窗＋cooldownMaxMs」者(久未觸發者n歸零, store不無限長)
//   與畸形紀錄(at非正數、n非正整數); 舊格式之數字讀入視為{ at, n: 1, reason: 'timeout' }(資料遷移)。
//   注意: 修前之遞補層讀到新格式之物件會當過期而刪除, 共用同一份持久store之所有使用者須同一次升版(README〈Breaking changes〉)。
//
// 【持久連敗】state.failStreak[id] = { k, at }: 組盡(本組未成交而試完)且實際送出之嘗試數>0、各次errorType不全為'params'時k+1, 成交刪除;
//   k達coolAfterFails即以reason 'streak'觸發冷卻; 觸發後不歸零(其後每次再敗皆重申, 視窗外即加長); 只在cooldownMs>0且coolAfterFails>0時計數;
//   at為最近一次遞增之時刻(供診斷; 不依時間過期, 掃表只刪畸形紀錄)。連敗存於store而非行程記憶體, 每輪一個新行程之呼叫端亦能累計。
//
// 【匯出】coolEntry／coolingOf／streakOf供呼叫端(如知識管線之免費池與健康摘要)以同一規則寫讀; 第一個參數可為state物件或store{get,set}(同forgetQuota)。
//   一般參數無效(state非物件、id非非空字串、cooldownMs非正整數、reason不在清單)即同步throw(K2, 先於任何副作用);
//   參數有效但寫不進(state凍結、store.get拋錯或回傳非物件、store.set拋錯)時回null、不throw。


//預設值
let DEFAULT_COOLDOWN_MAX_MS = 24 * 60 * 60 * 1000 //24小時(主控裁示D13甲; 原則1: 未給不得等於不限)
let DEFAULT_GROWTH = 1


//原因清單與固定視窗原因
let REASONS = ['timeout', 'rate-limit', 'detect', 'streak', 'quota', 'budget']
let FIXED_REASONS = new Set(['quota', 'budget'])


/**
 * 冷卻設定之正規化(非預期之值逐鍵改用預設, K3)
 *
 * @param {Object} o 輸入{ cooldownMs, cooldownGrowth, cooldownMaxMs }
 * @returns {Object} 回傳{ cooldownMs(非負整數，0代表未啟用), growth(≥1之有限數), maxMs(≥cooldownMs之正整數) }
 */
function coolCfgOf(o) {
    let cooldownMs = ispint(o?.cooldownMs) ? cint(o.cooldownMs) : 0
    let g = o?.cooldownGrowth
    let gn = isenum(g) ? Number(typeof g === 'string' ? g.trim() : g) : NaN //數字類收數字字串(K7)
    let growth = (Number.isFinite(gn) && gn >= 1) ? gn : DEFAULT_GROWTH
    let maxMs = ispint(o?.cooldownMaxMs) ? cint(o.cooldownMaxMs) : DEFAULT_COOLDOWN_MAX_MS
    if (maxMs < cooldownMs) {
        maxMs = cooldownMs
    }
    return { cooldownMs, growth, maxMs }
}


/**
 * 冷卻紀錄之正規化(舊格式數字→{at,n:1,reason:'timeout'}；at晚於now者改為now；畸形者回null)
 *
 * @param {*} v 輸入state.cooling[id]之值
 * @param {Number} now 輸入現在時刻毫秒
 * @returns {Object|null} 回傳{ at, n, reason }或null(畸形)
 */
function normRec(v, now) {
    let at = null
    let n = 1
    let reason = 'timeout'
    if (typeof v === 'number') {
        if (!(Number.isFinite(v) && v > 0)) {
            return null
        }
        at = v
    }
    else if (isobj(v)) {
        let a = v.at
        let k = v.n
        if (!(typeof a === 'number' && Number.isFinite(a) && a > 0)) {
            return null
        }
        if (!(Number.isInteger(k) && k >= 1)) {
            return null
        }
        at = a
        n = k
        reason = REASONS.includes(v.reason) ? v.reason : 'timeout'
    }
    else {
        return null
    }
    if (at > now) {
        at = now //防時鐘回撥: 未來時刻視為現在
    }
    return { at, n, reason }
}


/**
 * 冷卻紀錄之有效視窗毫秒
 *
 * @param {Object} rec 輸入正規化之紀錄{ at, n, reason }
 * @param {Object} cfg 輸入coolCfgOf之結果
 * @returns {Number} 回傳有效視窗毫秒
 */
function windowOf(rec, cfg) {
    if (FIXED_REASONS.has(rec.reason)) {
        return cfg.cooldownMs
    }
    let w = cfg.cooldownMs * Math.pow(cfg.growth, rec.n - 1)
    if (!Number.isFinite(w) || w > cfg.maxMs) {
        return cfg.maxMs
    }
    return Math.floor(w)
}


/**
 * 紀錄是否仍在冷卻視窗內
 *
 * @param {Object} rec 輸入正規化之紀錄
 * @param {Object} cfg 輸入coolCfgOf之結果
 * @param {Number} now 輸入現在時刻毫秒
 * @returns {Boolean} 回傳是否冷卻中
 */
function isCoolingRec(rec, cfg, now) {
    return (now - rec.at) < windowOf(rec, cfg)
}


/**
 * 掃整張冷卻表與連敗表：舊格式改寫為新格式、未來時刻改為現在、畸形紀錄刪除、久未觸發者清除；有改動回true(呼叫端寫回)
 *
 * @param {Object} state 輸入狀態物件(cooling必為物件；failStreak可無)
 * @param {Object} cfg 輸入coolCfgOf之結果
 * @param {Number} now 輸入現在時刻毫秒
 * @returns {Boolean} 回傳是否有改動
 */
function sweepCooling(state, cfg, now) {
    let dirty = false
    try {
        let table = state.cooling
        for (let id of Object.keys(table)) {
            let v = table[id]
            let rec = normRec(v, now)
            if (rec === null || (now - rec.at) >= windowOf(rec, cfg) + cfg.maxMs) {
                delete table[id]
                dirty = true
                continue
            }
            if (!isobj(v) || v.at !== rec.at || v.reason !== rec.reason) {
                table[id] = rec
                dirty = true
            }
        }
        //持久連敗: 只刪畸形紀錄; 不依時間過期(設計R-COOL-3: 跨輪累計、觸發後不歸零、成交歸零, 無過期規則——依cooldownMaxMs過期會令短上限＋長輪距之呼叫端永遠累計不到門檻, 外審terra F05)
        let fs = state.failStreak
        if (isobj(fs)) {
            for (let id of Object.keys(fs)) {
                let s = fs[id]
                let ok = isobj(s) && Number.isInteger(s.k) && s.k >= 1 && typeof s.at === 'number' && Number.isFinite(s.at)
                if (!ok) {
                    delete fs[id]
                    dirty = true
                }
            }
        }
    }
    catch {} //凍結之state: 不清理(退化為修前之讀取行為)
    return dirty
}


/**
 * 寫入一次冷卻觸發(視窗內只更新at；視窗外依原因決定n是否+1；無紀錄則新建n為1)
 *
 * @param {Object} state 輸入狀態物件
 * @param {String} id 輸入條目id
 * @param {String} reason 輸入原因(REASONS之一)
 * @param {Object} cfg 輸入coolCfgOf之結果
 * @param {Number} now 輸入現在時刻毫秒
 * @param {Boolean} [noInc=false] 輸入是否不得+1(遞補層於同一次呼叫內已為該條目+1過時傳true，令每次呼叫每條目至多+1)
 * @returns {Object|null} 回傳{ rec(寫入之紀錄), cooldownMs(有效視窗), n0(寫入前之n，無紀錄為0), r0(寫入前之reason，無紀錄為null), incremented(是否+1), renewed(寫入前已在有效視窗內＝視窗內之重申，只更新at；新建或視窗外為false) }，寫不進(凍結)回null
 */
function applyCool(state, id, reason, cfg, now, noInc = false) {
    try {
        if (!isobj(state.cooling)) {
            state.cooling = {}
        }
        let cur = normRec(state.cooling[id], now)
        let rec = null
        let incremented = false
        let renewed = cur !== null && isCoolingRec(cur, cfg, now) //寫入前是否已在有效視窗內(第421項; 與noInc無關, 只看寫入前之紀錄)
        if (cur === null) {
            rec = { at: now, n: 1, reason }
        }
        else if (noInc === true || isCoolingRec(cur, cfg, now)) {
            rec = { at: now, n: cur.n, reason: cur.reason } //視窗內: 只更新at
        }
        else {
            let n = FIXED_REASONS.has(reason) ? cur.n : cur.n + 1
            incremented = n > cur.n
            rec = { at: now, n, reason }
        }
        state.cooling[id] = rec
        return { rec, cooldownMs: windowOf(rec, cfg), n0: cur === null ? 0 : cur.n, r0: cur === null ? null : cur.reason, incremented, renewed }
    }
    catch {
        return null
    }
}


/**
 * 連敗+1(無紀錄則建k為1)
 *
 * @param {Object} state 輸入狀態物件
 * @param {String} id 輸入條目id
 * @param {Number} now 輸入現在時刻毫秒
 * @returns {Object|null} 回傳{ k(遞增後), k0(遞增前，無紀錄為0) }，寫不進回null
 */
function incStreak(state, id, now) {
    try {
        if (!isobj(state.failStreak)) {
            state.failStreak = {}
        }
        let s = state.failStreak[id]
        let k0 = (isobj(s) && Number.isInteger(s.k) && s.k >= 1) ? s.k : 0
        state.failStreak[id] = { k: k0 + 1, at: now }
        return { k: k0 + 1, k0 }
    }
    catch {
        return null
    }
}


/**
 * 連敗歸零(刪紀錄)
 *
 * @param {Object} state 輸入狀態物件
 * @param {String} id 輸入條目id
 * @returns {Boolean} 回傳是否有刪除
 */
function resetStreak(state, id) {
    try {
        if (isobj(state.failStreak) && state.failStreak[id] !== undefined) {
            delete state.failStreak[id]
            return true
        }
    }
    catch {}
    return false
}


/**
 * 取現在時刻(opt.now可為函數或數字，非預期者用Date.now)
 *
 * @param {*} now 輸入opt.now
 * @returns {Number} 回傳毫秒
 */
function nowOf(now) {
    if (isfun(now)) {
        let h = callHookSync(() => now())
        if (h.state === 'value' && typeof h.value === 'number' && Number.isFinite(h.value)) {
            return h.value
        }
        return Date.now()
    }
    if (typeof now === 'number' && Number.isFinite(now)) {
        return now
    }
    return Date.now()
}


/**
 * 檢測第一個參數(state物件或store)與id；無效同步throw
 *
 * @param {String} fn 輸入函數名
 * @param {*} target 輸入state或store
 * @param {*} id 輸入條目id
 * @returns {Boolean} 回傳是否為store
 */
function checkTarget(fn, target, id) {
    let kind = callHookSync(() => ({ obj: isobj(target), store: isobj(target) && isfun(target.get) && isfun(target.set) }))
    if (kind.state !== 'value' || kind.value.obj !== true) {
        throw new Error(`${fn}: invalid state: must be a state object or a store with get and set`)
    }
    if (!isestr(id)) {
        throw new Error(`${fn}: invalid id: must be a non-empty string`)
    }
    return kind.value.store === true
}


/**
 * 讀出state(store者經get；拋錯、回傳Promise或非物件回null)
 *
 * @param {*} target 輸入state或store
 * @param {Boolean} isStore 輸入是否為store
 * @returns {Object|null} 回傳state物件或null
 */
function readState(target, isStore) {
    if (!isStore) {
        return target
    }
    let g = callHookSync(() => target.get())
    if (g.state !== 'value' || !isobj(g.value)) {
        return null
    }
    return g.value
}


/**
 * 以遞補層之同一規則寫入一次條目冷卻(供呼叫端之額度池等寫入者使用，不另寫時間戳)
 *
 * 規則：視窗內之觸發只更新at；視窗外之觸發，reason為'quota'或'budget'(固定視窗原因)時不動n，其餘n+1；無紀錄則新建n為1；
 * 有效視窗＝固定視窗原因 ? cooldownMs : min(cooldownMs × cooldownGrowth^(n−1), cooldownMaxMs)(詳見本檔檔頭)。
 * 第一個參數可為state物件或store{get,set}(經get讀出、寫入後set寫回一次，同forgetQuota)。
 * state非物件或不可讀、id非非空字串、cooldownMs非正整數、reason不在清單時同步throw(先於任何副作用)；寫不進(state凍結、store.get拋錯或回傳非物件、store.set拋錯)回null
 *
 * @param {Object} state 輸入遞補層狀態物件或store{get,set}
 * @param {String} id 輸入條目id
 * @param {Object} opt 輸入設定物件
 * @param {Number} opt.cooldownMs 輸入冷卻視窗毫秒正整數(同dispatchAiFallback之cooldownMs)
 * @param {String} [opt.reason='timeout'] 輸入原因字串：'timeout'、'rate-limit'、'detect'、'streak'、'quota'、'budget'，預設'timeout'
 * @param {Number} [opt.cooldownGrowth=1] 輸入視窗外再觸發時之視窗倍率(≥1)，非預期之值用1
 * @param {Number} [opt.cooldownMaxMs=86400000] 輸入加長之上限毫秒正整數(小於cooldownMs取cooldownMs)，非預期之值用24小時
 * @param {Number|Function} [opt.now=Date.now] 輸入現在時刻毫秒或回傳之函數，預設Date.now
 * @returns {Object|null} 回傳{ at, n, reason, cooldownMs(本次有效視窗), until(at＋有效視窗), renewed(寫入前已在有效視窗內＝視窗內之重申〔只更新at，n與reason不變〕為true；新建或視窗外之新觸發為false) }，寫不進回null
 * @example
 *
 * import { coolEntry } from './src/cooling.mjs'
 *
 * let state = { cursors: {}, cooling: {} }
 * console.log(coolEntry(state, 'oc:opencode/beam-free', { reason: 'quota', cooldownMs: 900000, now: 1000 }))
 * // => { at: 1000, n: 1, reason: 'quota', cooldownMs: 900000, until: 901000, renewed: false }
 *
 * console.log(coolEntry(state, 'oc:opencode/beam-free', { reason: 'quota', cooldownMs: 900000, now: 2000 }))
 * // => { at: 2000, n: 1, reason: 'quota', cooldownMs: 900000, until: 902000, renewed: true }
 *
 */
function coolEntry(state, id, opt = {}) {
    let isStore = checkTarget('coolEntry', state, id)
    if (!ispint(opt?.cooldownMs)) {
        throw new Error('coolEntry: invalid opt.cooldownMs: must be a positive integer')
    }
    let reason = (opt.reason === undefined || opt.reason === null) ? 'timeout' : opt.reason
    if (!REASONS.includes(reason)) {
        throw new Error(`coolEntry: invalid opt.reason: must be one of ${REASONS.join(', ')}`)
    }
    let cfg = coolCfgOf(opt)
    let now = nowOf(opt.now)
    let st = readState(state, isStore)
    if (st === null) {
        return null
    }
    let w = applyCool(st, id, reason, cfg, now)
    if (w === null) {
        return null
    }
    if (isStore) {
        let s = callHookSync(() => state.set(st))
        if (s.state === 'threw') {
            return null
        }
    }
    return { ...w.rec, cooldownMs: w.cooldownMs, until: w.rec.at + w.cooldownMs, renewed: w.renewed }
}


/**
 * 讀出條目之冷卻紀錄(唯讀，不改寫store；舊格式數字以{ n: 1, reason: 'timeout' }呈現)
 *
 * 給opt.cooldownMs(正整數)時另回本紀錄之有效視窗cooldownMs、until與cooling(現在是否仍在視窗內)；未給則只回紀錄本身。
 * state非物件或不可讀、id非非空字串時同步throw；無紀錄、紀錄畸形、store讀不出時回null
 *
 * @param {Object} state 輸入遞補層狀態物件或store{get,set}
 * @param {String} id 輸入條目id
 * @param {Object} [opt={}] 輸入設定物件{ cooldownMs, cooldownGrowth, cooldownMaxMs, now }，同coolEntry
 * @returns {Object|null} 回傳{ at, n, reason }(給cooldownMs時另帶cooldownMs、until、cooling)，或null
 * @example
 *
 * import { coolingOf } from './src/cooling.mjs'
 *
 * let state = { cooling: { 'a': { at: 1000, n: 2, reason: 'timeout' } } }
 * console.log(coolingOf(state, 'a', { cooldownMs: 900000, cooldownGrowth: 4, now: 2000 }))
 * // => { at: 1000, n: 2, reason: 'timeout', cooldownMs: 3600000, until: 3601000, cooling: true }
 *
 */
function coolingOf(state, id, opt = {}) {
    let isStore = checkTarget('coolingOf', state, id)
    let st = readState(state, isStore)
    if (st === null) {
        return null
    }
    let now = nowOf(opt?.now)
    let rec = null
    try {
        rec = isobj(st.cooling) ? normRec(st.cooling[id], now) : null
    }
    catch {
        rec = null
    }
    if (rec === null) {
        return null
    }
    if (!ispint(opt?.cooldownMs)) {
        return rec
    }
    let cfg = coolCfgOf(opt)
    let w = windowOf(rec, cfg)
    return { ...rec, cooldownMs: w, until: rec.at + w, cooling: (now - rec.at) < w }
}


/**
 * 讀出條目之持久連敗次數(無紀錄回0)
 *
 * state非物件或不可讀、id非非空字串時同步throw；store讀不出時回0
 *
 * @param {Object} state 輸入遞補層狀態物件或store{get,set}
 * @param {String} id 輸入條目id
 * @returns {Number} 回傳連敗次數非負整數
 * @example
 *
 * import { streakOf } from './src/cooling.mjs'
 *
 * console.log(streakOf({ failStreak: { 'a': { k: 3, at: 1000 } } }, 'a'))
 * // => 3
 *
 */
function streakOf(state, id) {
    let isStore = checkTarget('streakOf', state, id)
    let st = readState(state, isStore)
    if (st === null) {
        return 0
    }
    try {
        let s = isobj(st.failStreak) ? st.failStreak[id] : null
        return (isobj(s) && Number.isInteger(s.k) && s.k >= 1) ? s.k : 0
    }
    catch {
        return 0
    }
}
