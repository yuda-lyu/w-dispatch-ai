//
// agy為Google官方之Gemini CLI後繼者(Gemini CLI已於2026-06-18停服), 命令名為agy非antigravity。
// 實測其models清單除Gemini系外亦含claude與gpt-oss等模型, 屬多模型閘道。
//
// 【提示詞兩種傳法: opt.promptVia(2026-11-07 Gitea #94; 帳本R-AGY-4)】預設'auto'。
//   'argv'  : prompt走`--print`旗標(命令列), 受Windows命令列上限約束(下節), 超過30000字元回params失敗; 行為同修前, 逐位元不變。
//   'stdin' : 走`--input-format stream-json`, 自stdin讀一行NDJSON `{"event":"user","message":{"role":"user","content":<prompt>}}`(JSON.stringify保證單行),
//             argv不帶`--print`、固定加`--input-format stream-json --output-format stream-json`; 不受命令列上限約束(agy 1.5.2實測37504字元成交)。
//   'auto'  : prompt不超過MAX_PROMPT_LENGTH走argv(與修前相同), 超過走stdin。
//   預設取'auto'而非'stdin'之理由: 本套件不安裝亦不鎖定各機agy版本(僅額度查詢有自己的版本把關), stdin路徑須agy>=1.1.15(見checkAgyStdinSupport.mjs);
//   預設'stdin'會令所有現行呼叫改走新路徑——舊版agy上原本可用者變成錯誤、extraArgs自帶之輸出旗標變成衝突、輸出來源由純文字改為NDJSON解析。
//   'auto'令既有呼叫(不超過上限者)零變更, 只有原本必然失敗之超長prompt變成可成交。要全面改走stdin者明給promptVia:'stdin'。
//
// 【stdin路徑之判成敗: 只看result事件, 不看離開碼與stdout是否非空(2026-11-07以agy 1.5.2實測)】
//   輸出為NDJSON(init、step_update…、末行result)。取最後一個result事件: status為SUCCESS → ok:true, stdout改為result.response(純回覆文字, 與argv路徑同一契約,
//   validate/nonempty/JSON解析皆對此判定), 原始NDJSON放結果之raw欄; status不為SUCCESS(如ERROR) → ok:false(errorType exec)。
//   **無result事件一律ok:false**: agy對不認得之輸入事件只在stderr印warning、stdout只留init事件、**離開碼0**(實測event為user_message), 離開碼0或非空stdout皆不得判成功。
//   validate在改寫stdout之後才判(經runExecCli之reshapeResult接縫, 先重整再驗證)。
//   錯誤輸入的其他形狀(實測): 無event欄 → exit 1「stream input message is missing the "event" field」; event為user但無message → exit 1; 帶`-p`又帶`--input-format` → exit 2(prompt不得同時由命令列給)。
//   extraArgs含`--output-format`、`--input-format`、`--print`/`-p`/`--prompt`、`-i`/`--prompt-interactive`(含`=值`與單橫線寫法)者, stdin路徑一律明確拒絕(params, 不spawn):
//   前兩者與本路徑固定旗標衝突(不依賴agy對重複旗標之last-wins行為, 未實測), 後三者agy會立即結束或進互動模式卡住。
//   不重試: 「無result事件」是在execCli回報成功後才判定, 不經execCli的maxRetries重試(多為確定性原因, 如不認得之輸入事件)。
//
// 【舊版agy → 明確錯誤】stdin路徑spawn前先以`agy --version`探測(約0.15秒, 同行程快取已確認支援者, 見checkAgyStdinSupport.mjs), 版本低於1.1.15直接回params錯誤並寫明須升級;
//   探測不確定(未能啟動/解析不出版本)時放行, 若agy回「not defined … input-format」之旗標錯誤, 結果之error改寫為同樣的升級指引。**輸入訊息形狀僅在1.5.2實測**, 1.1.15～1.5.1未驗證。
//
// 【argv路徑: prompt走--print旗標(2026-08-10於本機agy 1.1.11實測確認)】
//   --print為帶值旗標(無值即回`flag needs an argument: -print`), 不可能由stdin取prompt;
//   把prompt塞stdin而args無--print會進互動模式卡住(stdin路徑須改用上節的stream-json, 不是把純文字塞stdin)。
//
// 【命令列長度上限(實測; 僅argv路徑)】prompt作為命令列參數受Windows CreateProcess之32767字元上限:
//   純文字32600可spawn、32700即spawn ENAMETOOLONG。含引號之prompt經spawn轉義後長度會膨脹
//   (實測2萬個引號轉義後仍炸), 故前置長度檢查只涵蓋常見情境(給清楚錯誤訊息, errorType params, attempts 0);
//   轉義膨脹等漏網情境由execCli回報: wsemi>=1.9.15起spawn同步拋錯不再reject, 而回結果物件
//   (ok:false、code:null、pid:undefined、attempts:1、error為'ENAMETOOLONG: spawn ENAMETOOLONG'),
//   經attachErrorType歸'spawn'(結構判定, 見getErrorType.mjs), 故本轉接器不再自包try/catch(與另三家CLI轉接器一致)。
//   stdin路徑不受此限(auto下含大量引號而長度未過上限者仍走argv, 仍可能炸; 要避開者明給promptVia:'stdin')。
//
// 【model與effort之衝突規則(實測)】帶檔位之slug(如gemini-4-argon-high)與--effort同時給定時,
//   檔位「不一致」才拒絕(`--model gemini-4-argon-high conflicts with --effort=low`, exit 1; 2026-10-16於agy 1.4.0重測),
//   一致(gemini-4-argon-high + --effort high)則放行; 基礎slug(如gemini-4-argon, 預設檔位medium)搭配effort亦放行。
//   本轉接器不預判slug格式(模型清單會演進), 衝突時由agy回報conflicts錯誤。
//
// 【未給model時由agy決定(2026-10-16以agy 1.4.0實測)】agy預設即Gemini 4 Argon (Medium)(日誌label與額度前後差皆證),
//   故未給model者會耗用Argon之7天額度; 內建providers條目皆明給model, 不受影響。預設由agy決定、可能隨版本變動, 本套件不維護。
//
// 【雙逾時:外層先到, agy自身之print-timeout只作後備(2026-10-16以agy 1.4.0實測改)】agy自身有--print-timeout(1.4.0之--help載明預設0s=等到完成),
//   與execCli之timeoutMs為兩層獨立逾時。agy自身逾時時**以離開碼0交回部分輸出**(stderr為「[agy] print timeout after Ns with turn in progress;
//   returning partial output」, stdout為已產出之部分、可能為空), 不是錯誤——原設計令它先於外層到時(舊推導max(30, timeoutMs秒−30), 自agy啟動後起算),
//   故自然耗時超過該截止點之回合一律被當成ok:true之截斷輸出(連超過timeoutMs者亦然), 遞補層之TIMEOUT分流與冷卻皆不可達(成功還會解除既有冷卻)。
//   現printTimeout未給時推導為timeoutMs加30秒(timeoutMs超過execCli計時器上限2147483647者同樣封頂), 令外層先逾時(error以TIMEOUT開頭);
//   agy之自身計時只在外層強殺失效時作後備。行為變更: 自然耗時介於舊截止點與timeoutMs者由截斷改為完整成功, 達timeoutMs者由「部分輸出ok:true」改為TIMEOUT失敗;
//   仍要部分輸出者明給小於timeoutMs之printTimeout(並自行驗證完整性)。
//
// 【權限拒絕(2026-10-16以agy 1.4.0實測)】skipPermissions為false時, 工具需要寫入權限而被擋下, agy以離開碼0回應、stdout為空,
//   stderr帶說明(`jetski: no output produced — a tool required the "write_file" permission ... auto-denied`); 須以空輸出判別, 不能只看ok。

/**
 * 找出extraArgs中與stdin路徑衝突之旗標(不分單雙橫線與`=值`寫法)
 *
 * @param {Array} extraArgs 輸入呼叫端額外命令列旗標，非陣列視為無
 * @returns {String} 回傳第一個衝突之原字串，無衝突為''
 */
function findStdinConflictArg(extraArgs) {
    for (let s of getCliArgs(extraArgs)) {
        let m = FLAG_NAME_RE.exec(s)
        if (m && STDIN_CONFLICT_FLAGS.includes(m[1])) {
            return s
        }
    }
    return ''
}


/**
 * 重整stdin路徑(stream-json)之execCli結果：成功時stdout改為result.response，無result事件或status非SUCCESS者判失敗，舊版agy不認得旗標者改寫為升級指引
 *
 * 此函數由runExecCli於驗證之前套用(reshapeResult接縫)，故validate判定的是改寫後之stdout
 *
 * @param {Object} r 輸入execCli之結果物件
 * @param {String} raw 輸入該次嘗試累積之完整stdout字串(失敗結果之stdout被execCli裁到500字元，故由onStdout自行累積；累積為空時取r.stdout)
 * @returns {Object} 回傳重整後之結果物件
 */
function reshapeAgyResult(r, raw) {
    let p = parseAgyStream(raw)
    let ok = get(r, 'ok', false) === true
    let stderr = String(get(r, 'stderr', '') || '')

    //execCli視為成功(離開碼0): 以result事件為準
    if (ok) {
        if (!p.hasResult) {
            let tail = stderr.trim().slice(-200)
            return {
                ...r,
                ok: false,
                stdout: '',
                raw,
                error: `agy stream-json: no result event in output (exit code ${get(r, 'code', 0)}); agy did not finish a turn for the input message (unsupported input event, print-timeout reached, or process ended early)${tail !== '' ? ` | stderr: ${tail}` : ''}`,
                errorType: 'exec',
            }
        }
        if (p.ok) {
            return { ...r, stdout: p.response, raw }
        }
        return {
            ...r,
            ok: false,
            stdout: strTruncate(p.response, 500, optTruncate),
            raw,
            error: `agy stream-json: result status ${p.status !== '' ? p.status : 'unknown'}${p.error !== '' ? `: ${p.error}` : ''}`,
            errorType: 'exec',
        }
    }

    //失敗(非零離開、逾時、未能啟動): 保留execCli之error與errorType判定, 僅整理stdout與補充診斷
    let out = { ...r }
    if (p.hasEvents) {
        out.stdout = p.hasResult ? strTruncate(p.response, 500, optTruncate) : ''
    }
    if (p.hasResult && !p.ok) {
        out.error = `${get(r, 'error', '')} | agy stream-json: result status ${p.status !== '' ? p.status : 'unknown'}${p.error !== '' ? `: ${p.error}` : ''}`
    }

    //舊版agy不認得--input-format(Go flag套件之未知旗標錯誤): 改寫為升級指引, 與探測低版本同一類(params, 與金鑰無關)
    if (!p.hasEvents && /not defined/i.test(stderr) && /input-format/i.test(stderr)) {
        out.error = `agy does not recognize --input-format (stdin prompt path needs agy >= ${MIN_STDIN_VERSION}); run [agy update], or use promptVia 'argv' for prompts within ${MAX_PROMPT_LENGTH} chars | stderr: ${stderr.trim().split(/\r?\n/)[0].slice(0, 200)}`
        out.errorType = 'params'
    }
    return out
}


/**
 * 以Google Antigravity CLI(agy)呼叫AI模型
 *
 * 特點：
 * prompt有兩種傳法(promptVia)：argv走`--print`旗標(命令列)，受命令列長度上限約束，超過30000字元回傳錯誤結果物件；
 * stdin走`--input-format stream-json`(agy>=1.1.15，輸入訊息形狀以1.5.2實測)，自stdin送一行NDJSON，不受命令列上限約束，輸出解析result事件(stdout仍為純回覆文字，原始NDJSON放raw，無result事件一律判失敗)；
 * 預設auto：不超過30000字元走argv(行為同修前)，超過走stdin，細節與預設值理由見檔頭；
 * model須為`agy models`第一欄之slug(如gemini-3.8-flash-low)，注意agy錯誤訊息列出的是顯示名稱而非slug；未給model時由agy決定(agy 1.4.0實測預設為Gemini 4 Argon (Medium)，耗用Argon額度)；
 * 帶檔位之slug(-high/-medium/-low結尾)與effort同時給定且檔位不一致時agy回conflicts錯誤(一致則放行)，
 * effort需agy>=1.1.11，建議搭配不帶檔位之基礎slug(如gemini-4-argon)使用；
 * 預設帶`--dangerously-skip-permissions`令非互動print模式不卡權限確認，可給予skipPermissions為false保留權限閘門(被擋下時agy以離開碼0回應、stdout為空)；
 * agy自身之--print-timeout未給時由timeoutMs加30秒推導，令外層先逾時(agy自身逾時是離開碼0之部分輸出而非錯誤，見檔頭)；
 * 沿用agy既有OAuth登入狀態(首次須於桌面互動模式完成登入)；
 * 本函數不會reject，一律以結果物件之ok與error欄位回報成敗
 *
 * @param {String} prompt 輸入提示詞字串，依promptVia作為`--print`旗標之值(argv，長度上限30000字元)或stdin之stream-json訊息內容(stdin，無長度上限)傳遞
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.promptVia='auto'] 輸入提示詞傳法字串，'argv'走`--print`旗標(超過30000字元回params失敗，行為同修前)，'stdin'走`--input-format stream-json`自stdin送一行NDJSON(不受命令列上限約束，需agy>=1.1.15；舊版回params錯誤並指引升級；extraArgs含`--output-format`、`--input-format`、`--print`/`-p`/`--prompt`、`-i`/`--prompt-interactive`者回params錯誤)，'auto'為不超過30000字元走argv、超過走stdin；其他值回退預設，預設'auto'
 * @param {String} [opt.exe='agy'] 輸入agy執行檔名稱或絕對路徑字串，命令名為agy非antigravity，給予名稱時由execCli自系統PATH解析(.cmd/.bat只給檔名時依cmd.exe語意先搜cwd再搜PATH，cwd不可信時請給絕對路徑)，預設'agy'
 * @param {String} [opt.model=''] 輸入模型slug字串，須為`agy models`第一欄之slug，例如'gemini-3.8-flash-low'、'gemini-4-argon-high'，預設''代表不帶`--model`旗標由agy自行決定(agy 1.4.0實測預設為Gemini 4 Argon (Medium)，會耗用Argon額度；預設可能隨agy版本變動)
 * @param {String} [opt.effort=''] 輸入推理深度字串，可選'low'、'medium'、'high'(部分模型更少，以`agy -p /effort --model <slug> --output-format json`之available為準)，需agy>=1.1.11，建議搭配不帶檔位之基礎slug；與帶檔位slug併用且檔位不一致時agy回conflicts錯誤，預設''代表不帶
 * @param {Boolean} [opt.skipPermissions=true] 輸入是否帶`--dangerously-skip-permissions`旗標布林值，false代表保留CLI權限閘門，預設true
 * @param {String} [opt.printTimeout=''] 輸入agy自身等待上限字串(如'10m'、'630s')，預設''代表由timeoutMs推導(加30秒，恆大於timeoutMs令外層先逾時；agy自身逾時是離開碼0之部分輸出，不是錯誤，故一般無須手動設)
 * @param {Array} [opt.addDirs=自動納入cwd] 輸入加入workspace之目錄字串陣列，逐項展開為`--add-dir`。agy以自身scratch目錄為工作區而不採子進程cwd，故未給本參數時自動納入有效cwd令檔案可視範圍與其他CLI轉接器一致；明示給陣列(含空陣列[]代表不揭露任何目錄)則完全尊重呼叫端
 * @param {Array} [opt.extraArgs=[]] 輸入額外命令列旗標字串陣列(如--output-format、--json-schema、--mode)，將接於固定旗標之後、`--print`(argv路徑)或`--input-format stream-json --output-format stream-json`(stdin路徑，此路徑下extraArgs不得含衝突旗標，見promptVia)之前，預設[]
 * @param {Number} [opt.timeoutMs=300000] 輸入逾時毫秒正整數，逾時將強制關閉子進程及其子孫程序，全套件統一預設300000
 * @param {String} [opt.cwd=process.cwd()] 輸入子進程工作目錄字串，預設process.cwd()。注意本參數不影響agy之檔案可視範圍(agy以自身scratch目錄為工作區)，可視範圍由addDirs決定(未給addDirs時自動納入本目錄)
 * @param {String|Function} [opt.validate=undefined] 輸入stdout驗證規則字串或自訂驗證函數，規則字串支援'nonempty'、'json'、'min:100'，多規則可用逗號串接，自訂函數可回傳Promise(以resolve值判定，與子進程共用該次嘗試之timeoutMs)，拋錯或reject視同驗證失敗(OUTPUT_VALIDATION_FAILED，stderr前綴'validate threw: 訊息')，逾時亦同(前綴'validate timed out: did not settle within Ns')，durationMs含驗證時間，預設undefined代表不驗證
 * @param {Number} [opt.maxRetries=0] 輸入失敗後最大重試次數非負整數，預設0
 * @param {Boolean|Object} [opt.spawnGate=undefined] 輸入spawn前之機器級守門設定：未給為預設(antigravity預設不限，守門為no-op、零成本；agy.exe之CPU占用本機未測)，false關閉，物件可明給maxCli(數agy.exe)、ramUsedMax、cpuMax等逐次開啟，等待不計入timeoutMs，見cliSpawnGate.mjs檔頭，預設undefined
 * @returns {Promise} 回傳Promise，resolve回傳結果物件，內含ok(是否成功布林值)、stdout(標準輸出字串，stdin路徑為result.response之純回覆文字)、stderr(標準錯誤字串，自訂validate拋錯時前綴'validate threw: 訊息')、code(離開碼，子進程未能啟動時為null)、error(錯誤訊息字串，成功時為空字串)、errorType(僅失敗時，機器可讀錯誤類別字串，子進程未能啟動為'spawn'，一覽見getErrorType.mjs檔頭)、durationMs(耗時毫秒)、pid(子進程pid，未能啟動時為undefined)、attempts(實際嘗試次數)；stdin路徑另帶raw(agy原始NDJSON輸出字串，離開碼0之結果才有)；失敗(離開碼非零、訊號結束、逾時)時另帶errorDetail({ exitCode(數字或null), stderrTail, stdoutTail(僅stdout非空) })，execCli原生形狀之error(`Exit code N`等)改為`Exit code N: <stderr尾段>`(至多300字元，已遮蔽金鑰權杖)，本轉接器自組之error(如`agy stream-json: …`)文字不變；成功時另帶finish(agy輸出不揭露終止原因故恆為'unknown')與outputChars(stdout字元數)，見runExecCli.mjs與failureDiag.mjs檔頭；本函數不會reject
 * @example
 * //need agy cli in system PATH, and OAuth login completed
 *
 * import dispatchAntigravity from './src/dispatchAntigravity.mjs'
 *
 * let test = async () => {
 *
 *     let r = await dispatchAntigravity('請只回覆兩個字：完成', { model: 'gemini-3.8-flash-low' })
 *     console.log(r.ok, r.stdout.trim())
 *     // => true 完成
 *
 *     //基礎slug搭配effort(不可用帶檔位之slug併用effort)
 *     let r2 = await dispatchAntigravity('請只回覆兩個字：完成', { model: 'gemini-4-argon', effort: 'low' })
 *     console.log(r2.ok)
 *     // => true
 *
 *     //超過30000字元(auto自動改走stdin; promptVia:'argv'則回prompt too long)
 *     let r3 = await dispatchAntigravity('填充文字。'.repeat(8000) + '請只回覆兩個字：完成', { model: 'gemini-3.8-flash-low' })
 *     console.log(r3.ok, r3.stdout.trim())
 *     // => true 完成
 *
 *     let re = await dispatchAntigravity('')
 *     console.log(re.ok, re.error)
 *     // => false 'prompt must be a non-empty string'
 *
 * }
 * await test()
 *     .catch((err) => {
 *         console.log(err)
 *     })
 *
 */
async function dispatchAntigravity(prompt, opt = {}) {

    //check prompt, 不reject故以錯誤結果物件回報
    prompt = asText(prompt) //String包裝物件取其原始字串(stdin只收原始字串, 空包裝不得繞過非空檢核); 判型不執行使用者程式碼(已撤銷之Proxy等回空字串)
    if (prompt === '') {
        return getErrorResult('prompt must be a non-empty string')
    }

    //promptVia, 無效值回退預設'auto'; viaStdin: stdin明給, 或auto且超過命令列上限(見檔頭【提示詞兩種傳法】)
    let promptVia = get(opt, 'promptVia', null)
    if (!PROMPT_VIA_LIST.includes(promptVia)) {
        promptVia = DEFAULT_PROMPT_VIA
    }
    let viaStdin = promptVia === 'stdin' || (promptVia === 'auto' && prompt.length > MAX_PROMPT_LENGTH)

    //check prompt length, 僅argv路徑: prompt走命令列旗標, 受CreateProcess 32767字元上限約束(stdin路徑不受此限)
    if (!viaStdin && prompt.length > MAX_PROMPT_LENGTH) {
        return getErrorResult(`prompt too long (${prompt.length} chars > ${MAX_PROMPT_LENGTH}), agy passes prompt via --print flag and Windows command line is capped at 32767 chars`)
    }

    //exe, 無效回退預設'agy', 由execCli自系統PATH解析實體路徑
    let exe = get(opt, 'exe', null)
    if (!isestr(exe)) {
        exe = DEFAULT_EXE
    }

    //model, 無效時整段`--model`旗標不出現, 由agy自行決定使用模型(agy 1.4.0實測預設為Gemini 4 Argon (Medium), 見檔頭)
    let model = get(opt, 'model', null)

    //effort, 無效時整段`--effort`旗標不出現; 與帶檔位slug之衝突由agy判定(檔位不一致時conflicts錯誤, exit 1)
    let effort = get(opt, 'effort', null)

    //skipPermissions, 非布林值回退預設true(非互動print模式不卡權限確認)
    let skipPermissions = get(opt, 'skipPermissions', null)
    if (!isbol(skipPermissions)) {
        skipPermissions = true
    }

    //timeoutMs, 先行取值以供printTimeout推導, 無效回退全套件統一預設300000
    let timeoutMs = castPintOr(get(opt, 'timeoutMs', null), DEFAULT_TIMEOUT_MS)

    //printTimeout, 未給時由timeoutMs加緩衝推導, 令外層先逾時(agy自身逾時為exit 0之部分輸出而非錯誤, 見檔頭)
    let printTimeout = get(opt, 'printTimeout', null)
    if (!isestr(printTimeout)) {
        printTimeout = `${Math.floor(Math.min(timeoutMs, MAX_TIMER_MS) / 1000) + PRINT_TIMEOUT_BUFFER_S}s`
    }

    //addDirs, 逐項展開為--add-dir(agy該旗標可重複)
    //未給時自動納入有效cwd(明給的或預設process.cwd()): agy以自身scratch目錄為工作區,
    //不採子進程之cwd, 僅--add-dir能擴其檔案可視範圍——僅給cwd時弱模型直接回「找不到檔案」、
    //強模型得自行摸索(實測116s vs 帶--add-dir僅6s), 且兩者皆ok:true屬靜默失敗,
    //極易被誤判為模型能力不足(2026-08-13以隨機token實測確認)。
    //明示addDirs(含空陣列[]代表不揭露任何目錄)則完全尊重呼叫端, 不自動加入
    let addDirs = get(opt, 'addDirs', null)
    if (!isarr(addDirs)) {
        let cwdEff = get(opt, 'cwd', null)
        if (!isestr(cwdEff)) {
            cwdEff = process.cwd()
        }
        addDirs = [cwdEff]
    }
    let addDirArgs = []
    for (let d of addDirs) {
        if (isestr(d)) {
            addDirArgs.push('--add-dir', d)
        }
    }

    //extraArgs
    let extraArgs = get(opt, 'extraArgs', null)

    //stdin路徑前置檢查(皆在spawn前、attempts 0): ①extraArgs不得含衝突旗標 ②agy版本須支援--input-format(見檔頭【舊版agy】)
    if (viaStdin) {
        let conflict = findStdinConflictArg(extraArgs)
        if (conflict !== '') {
            return getErrorResult(`extraArgs contains [${conflict}], which conflicts with the stdin prompt path (promptVia '${promptVia}' uses --input-format stream-json --output-format stream-json and no --print); remove it, or use promptVia 'argv' for prompts within ${MAX_PROMPT_LENGTH} chars`)
        }
        let sup = await checkAgyStdinSupport(exe, { cwd: get(opt, 'cwd', null), timeoutMs })
        if (sup.supported === false) {
            return getErrorResult(`agy version [${sup.version}] does not support --input-format stream-json (stdin prompt path needs agy >= ${sup.minVersion}); run [agy update], or use promptVia 'argv' for prompts within ${MAX_PROMPT_LENGTH} chars`)
        }
    }

    //args, argv路徑: prompt作為--print之值放最後; stdin路徑: 不帶--print, 固定旗標放最後(prompt由stdin送入)
    let args = getCliArgs(
        skipPermissions ? '--dangerously-skip-permissions' : [],
        ['--print-timeout', printTimeout],
        isestr(model) ? ['--model', model] : [],
        isestr(effort) ? ['--effort', effort] : [],
        addDirArgs,
        extraArgs,
        viaStdin ? ['--input-format', 'stream-json', '--output-format', 'stream-json'] : ['--print', prompt],
    )

    //optCli, 剔除本轉接器自用鍵後原樣轉傳, 令呼叫端可用execCli全部設定(例如onStdout、maxBuffer)
    let optCli = omit(opt, OWN_KEYS)

    //stdin路徑: input為一行NDJSON(JSON.stringify不輸出裸換行, 含換行與引號之prompt亦保持單行); 輸出經reshapeResult接縫還原
    //(onStdout累積全文: 失敗結果之stdout被execCli裁到500字元, result事件在最後一行會被裁掉; 作法同dispatchOpencode之detectTruncation)
    let input
    if (viaStdin) {
        input = `${JSON.stringify({ event: 'user', message: { role: 'user', content: prompt } })}\n`
        let acc = ''
        let userOnStdout = get(optCli, 'onStdout', null)
        optCli.onStdout = (str) => {
            if (acc.length < ACC_MAX) {
                acc += str
            }
            if (isfun(userOnStdout)) {
                userOnStdout(str) //呼叫端回調拋錯由execCli的callUser承接
            }
        }
        optCli.reshapeResult = (r) => {
            let raw = acc !== '' ? acc : String(get(r, 'stdout', '') || '')
            acc = '' //下一次嘗試重新累積
            return reshapeAgyResult(r, raw)
        }
    }

    //execCli, argv路徑不帶input(agy由--print取得prompt); 前置檢查漏網之spawn同步錯誤(如轉義膨脹之ENAMETOOLONG)
    //由execCli以結果物件回報(wsemi>=1.9.15, 不reject, 見檔頭); 經runExecCli: 自訂validate拋錯視同驗證失敗、失敗結果補上errorType(見runExecCli.mjs)
    //spawnGate(使用端給之守門設定, 原樣留在optCli)與spawnGateKind交runExecCli於每次spawn前守門(見cliSpawnGate.mjs; antigravity預設不限)
    return runExecCli(exe, args, {
        ...optCli,
        ...(viaStdin ? { input } : {}),
        timeoutMs,
        spawnGateKind: 'antigravity',
    })
}


export default dispatchAntigravity
